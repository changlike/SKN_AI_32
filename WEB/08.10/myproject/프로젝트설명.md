# 11. 정적 파일 설정

# 14. settings.py 에 python-dotenv 적용
```text
# 파일 경로를 편리하게 처리하기 위함
from pathlib import Path

# 운영체제의 환경변수를 읽기 위함
import os

# .env 파일의 내용을 환경변수로 읽음
from dotenv import load_dotenv

# Build paths inside the project like this: BASE_DIR / 'subdir'.
# 현재 프로젝트의 최상위 디렉토리를 구함 (프로젝트 루트임)
BASE_DIR = Path(__file__).resolve().parent.parent

# 프로젝트 루트의 .env 파일 읽음
load_dotenv(BASE_DIR / ".env")

# SECURITY 키 환경변수 적용
SECRET_KEY = os.getenv("DJANGO_SECRET_KEY")

# DEBUG 환경 변수 적용
DEBUG = (os.getenv("DJANGO_DEBUG", "True").lower() == "true")

ALLOWED_HOSTS = [
    host.strip() for host in os.getenv("DJANGO_ALLOWED_HOSTS", "127.0.0.1,localhost").split(",") if host.strip()
]
```

```text
-- Django 프로젝트가 사용할 db 생성

SHOW DATABASES;

-- django 전용 사용자계정 만듦
CREATE USER IF NOT EXISTS
'django_user'@'localhost'
IDENTIFIED BY 'django1234!';

-- 127.0.0.1 연결용도 만듦
CREATE USER IF NOT EXISTS
'django_user'@'127.0.0.1'
IDENTIFIED BY 'django1234!';

-- 권한을 부여함
GRANT ALL PRIVILEGES
ON django_member_board.*
TO 'django_user'@'localhost';

GRANT ALL PRIVILEGES
ON django_member_board.*
TO 'django_user'@'127.0.0.1';

-- 적용
FLUSH PRIVILEGES;
```

프로젝트 루트/db/create_database.sql 로 저장

# 16. Django 와 MySQL 연결
settings.py 에서
```text
DATABASES = {
    "default": {
        "ENGINE": "django.db.backends.mysql",
        
        # .env 의 DB 이름을 읽어옴
        "NAME": os.getenv("DB_NAME"),        
        # MySQL 사용자계정 등록
        "USER": os.getenv("DB_USER"),        
        # MySQL 비밀번호
        "PASSWORD": os.getenv("DB_PASSWORD"),        
        # MySQL 서버 주소
        "HOST": os.getenv("DB_HOST"),        
        # MySQL 포트
        "PORT": os.getenv("DB_PORT"),
        # 한글과 이모지 저장을 위해 utf8mb4 저장
        "OPTIONS": {
            "charset": "utf8mb4",
        }
    }
}
```
